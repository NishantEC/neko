  else if (mode === 'scan') { g.tx = 2.8 * Math.sin(t * 1.9 + i.ph[1]); g.ty = .5; }

      case 'read': g.tx = rand(-2.2, 2.2); g.ty = rand(1, 1.8); g.next = t + rand(.25, .7); break;
      case 'read-slow': g.tx = rand(-1.8, 1.8); g.ty = rand(.6, 1.4); g.next = t + rand(.7, 1.6); break;
      case 'user': g.tx = rand(-.5, .5); g.ty = .5; g.next = t + rand(1.6, 3.2); break;
      case 'still': g.tx = -1.1; g.ty = 1.3; g.next = t + 99; break;
      case 'down': g.tx = 0; g.ty = 1.5; g.next = t + 99; break;
      default: { const c = Math.random() < .4; g.tx = c ? 0 : rand(-2.2, 2.2); g.ty = c ? 0 : rand(-1.1, 1.1); g.next = t + rand(1.8, 5.2); }

function step(i, st, t, dt, red, force) {
  const T = i.target, p = i.p, k = 1 - Math.exp(-dt * 3);
  for (const key of NUM_KEYS) p[key] += (T[key] - p[key]) * (force ? 1 : k);
  const f = Math.max(.02, p.freq), o = i.o;
  o[0] += TAU * f * dt; o[1] += TAU * f * 1.618 * dt; o[2] += TAU * f * .731 * dt; o[3] += TAU * f * .9 * dt; o[4] += TAU * f * .613 * dt;
  let amp = p.amp, br = p.breath, tl = p.tilt, jit = p.jitter;
  if (red) amp = br = tl = jit = 0;
  if (i.calm) { amp *= .35; br *= .4; tl *= .35; jit = 0; }
  const lively = i.play && !red && st.presence !== 'blocked' && st.presence !== 'offline';
  if (lively) { amp = amp * 2 + .012; br = br * 1.7 + .006; tl = tl * 1.6 + .6; }
  if (lively && t > i.anticAt) { antic(i, t); i.anticAt = t + rand(4.5, 8.5); }
  if (i.track) { const on = !red && i.trk && t - i.trk.t < 2.6; const want = on ? clamp(i.trk.dx / 320, -1, 1) * 5 : 0; i.lean += (want - i.lean) * (1 - Math.exp(-dt * 5)); }
  const fy = amp * (Math.sin(o[0] + i.ph[0]) * .72 + Math.sin(o[1] + i.ph[1]) * .28);
  const fx = amp * .3 * Math.sin(o[2] + i.ph[2]);
  const b = br * Math.sin(o[3] + i.ph[3]);
  const j = jit ? jit * (Math.sin(t * 41 + i.ph[4]) * .6 + Math.sin(t * 67 + i.ph[5]) * .4) : 0;
  const rot = p.tiltBias + tl * Math.sin(o[4] + i.ph[4]);
  if (!red && !i.calm && T.hop && t > i.hopAt) { i.jy.v -= T.hopK; i.sq.v -= T.hopK; i.hopAt = t + rand(T.hop[0], T.hop[1]); }
  if (!T.hop && i.hopAt < t) i.hopAt = t + rand(2, 5);
  spring(i.jy, 170, 12, dt); spring(i.sq, 420, 16, dt); spring(i.tk, 200, 13, dt);
  const kk = i.calm ? .3 : 1, sq = i.sq.x * kk, px = i.px;
  i.body.style.transform = `translate3d(${r2((fx + j * .5) * px)}px,${r2((fy + i.jy.x * kk + j) * px)}px,0) rotate(${r2(rot + i.tk.x * kk + i.lean)}deg) scale(${(1 + b * .6 + sq).toFixed(4)},${(1 + b - sq).toFixed(4)})`;
  const name = t < i.ovUntil ? i.ov : faceFor(i.id, st);
  if (name !== i.exprName) {
    const tgt = EXPR[name] || EXPR.neutral;
    if (!i.e || red) { i.e = {...tgt}; i.eT = 1; } else { i.eFrom = {...i.e}; i.eTo = tgt; i.eT = 0; }
    i.exprName = name; i.dirty = true;
  }
  if (i.eT < 1) { i.eT = Math.min(1, i.eT + dt / .26); const u = easeBack(i.eT); for (const key of EXPR_KEYS) i.e[key] = lerp(i.eFrom[key], i.eTo[key], u); i.dirty = true; }
  const e = i.e;
  let bl = 1;
  if (!red && T.blink && t >= i.blinkAt) {
    i.blinkT = t;
    if (i.blinkQ > 0) { i.blinkQ--; i.blinkAt = t + .26; } else { i.blinkAt = t + rand(T.blink[0], T.blink[1]); if (Math.random() < .15) i.blinkQ = 1; }
  }
  const bu = (t - i.blinkT) / .16;

  const big = px >= 48;

  i.dirty = false;
  const gx = i.g.x, gy = i.g.y, lx = 50 - e.eg / 2 + gx, rx = 50 + e.eg / 2 + gx, ey = e.ey + gy, bs = bl.toFixed(3);
  const wu = (t - i.winkT) / .4, wl = i.play && wu >= 0 && wu < 1 ? 1 - Math.sin(wu * Math.PI) * .92 : 1;
  i.eyeL.setAttribute('transform', `translate(${r2(lx)} ${r2(ey)}) rotate(${r2(e.er)}) scale(1 ${(bl * wl).toFixed(3)})`);

}

    const st = stateOf(i);
    applyState(i, st, false);
    if (i.visible) step(i, st, t, dt, M.reduced, false);
  }
  for (const h of M.hooks) h(t, dt);

function hl(src) {
  const re = /(\/\/.*$|\{\/\*.*?\*\/\})|("[^"]*"|'[^']*'|`[^`]*`)|(<\/?[A-Za-z][\w.]*|\/?>)|\b(import|from|export|default|function|return|const|let|type|await|async)\b|([A-Za-z_][\w-]*)(?==)|\b(\d+)\b|([A-Z][A-Za-z]+)(?=[:(])|([{}()[\];,.:])/g;
  let out = '', last = 0, m;
  while ((m = re.exec(src))) {
    out += esc(src.slice(last, m.index));
    const cls = m[1] ? 'c' : m[2] ? 's' : m[3] ? 't' : m[4] ? 'k' : m[5] ? 'a' : m[6] ? 'n' : m[7] ? 'f' : 'p';
    out += `<span class="tk-${cls}">${esc(m[0])}</span>`;
    last = m.index + m[0].length;
  }
  return out + esc(src.slice(last));
}

  };

function inlineWords(parent, text) {
  const out = [];
  text.split(/(\*\*[^*]+\*\*|`[^`]+`)/).forEach(chunk => {
    if (!chunk) return;
    let host = parent, body = chunk;
    if (chunk.startsWith('**')) { host = parent.appendChild(document.createElement('strong')); body = chunk.slice(2, -2); }
    else if (chunk.startsWith('`')) { const c = parent.appendChild(document.createElement('code')); c.className = 'w'; c.textContent = chunk.slice(1, -1); out.push(c); return; }
    for (const w of body.split(/(\s+)/)) {
      if (!w) continue;
      if (/^\s+$/.test(w)) { host.appendChild(document.createTextNode(' ')); continue; }
      const s = document.createElement('span'); s.className = 'w'; s.textContent = w; host.appendChild(s); out.push(s);
    }
  });
  return out;
}
function buildWords(box, blocks) {
  const words = [];
  for (const b of blocks) {
    if (b.p) words.push(...inlineWords(box.appendChild(document.createElement('p')), b.p));
    if (b.ol) { const ol = box.appendChild(document.createElement('ol')); for (const it of b.ol) words.push(...inlineWords(ol.appendChild(document.createElement('li')), it)); }
    if (b.code) { const c = document.createElement('div'); c.className = 'cb w'; c.innerHTML = `<div class="cb-h">${ic('code')}<span>${esc(b.file)}</span><span class="sp"></span><button type="button" class="ib" aria-label="Copy code" data-copy>${ic('copy')}</button></div><pre>${b.code.split('\n').map(hl).join('\n')}</pre>`; box.appendChild(c); words.push(c); }
    if (b.refs) { const r = document.createElement('div'); r.className = 'refs w'; r.innerHTML = b.refs; box.appendChild(r); words.push(r); }
  }
  return words;
}

async function streamReply(id, blocks, o = {}) {
  const tok = ++S.chatRun, alive = () => tok === S.chatRun;
  if (activeTab !== 'chat') tabState('chat', {dirty:true, ac:id});
  const m = agentMsg(id);
  const drive = o.drive !== false && S.agents[id].presence === 'available';
  if (drive) setAgent(id, {presence:'thinking', label:`${A[id].name} is thinking…`});
  await sleep(o.think ?? 1100);
  if (alive() && o.tool) {
    const [icn, label, meta] = o.tool;
    const running = o.toolRunning || o.tool[3] || label;
    const run = toolCall(m, icn, running + '…', '', o.toolBody || `<p>${esc(meta)}</p>`, true);
    if (drive) setAgent(id, {presence:'tool', tool:A[id].tool, label: running});
    await sleep(o.toolMs ?? 1500);
    run.done(label, meta);
  }
  m.el.classList.add('answering');
  if (drive && alive()) setAgent(id, {presence:'working', label:'Writing a reply…'});
  const words = buildWords(m.text, blocks);
  const caret = document.createElement('span'); caret.className = 'caret';
  let n = 0;
  while (n < words.length && alive()) {
    const chunk = 1 + (Math.random() * 3 | 0);
    for (let k = 0; k < chunk && n < words.length; k++) words[n++].classList.add('on');
    const lw = words[n - 1];
    if (lw.tagName === 'SPAN' || lw.tagName === 'CODE') lw.after(caret); else caret.remove();
    chatScroll();
    await sleep(M.reduced ? 25 : rand(35, 85));
  }
  words.forEach(w => w.classList.add('on')); caret.remove();
  if (drive && ACTIVE.has(S.agents[id].presence)) setAgent(id, {presence:'available'});
  chatScroll();
}

function orbitHook(t, dt) {
  if (activeTab !== 'home' || !ORBIT.els.length) return;
  const st = $('#hero-stage'); if (!st) return;
  const w = st.clientWidth, h = st.clientHeight, cx = w / 2, cy = h / 2;
  if (!w) return;
  const rx = Math.min(w * .38, 190), ry = Math.min(48, h * .25);
  if (w !== ORBIT.w) { ORBIT.w = w; $('#hero-ring').setAttribute('viewBox', `0 0 ${w} ${h}`); const el = $('#hero-ell'); el.setAttribute('cx', cx); el.setAttribute('cy', cy); el.setAttribute('rx', rx); el.setAttribute('ry', ry); }
  if (!M.reduced) ORBIT.a += dt * .11 * (ORBIT.hover ? .3 : 1);
  ORBIT.els.forEach((el, k) => {
    const th = ORBIT.a + k * TAU / ORBIT.els.length + .5;
    const x = cx + rx * Math.cos(th), y = cy + ry * Math.sin(th), d = (Math.sin(th) + 1) / 2;
    el.style.transform = `translate(${r2(x)}px,${r2(y)}px) translate(-50%,-50%) scale(${(.74 + .32 * d).toFixed(3)})`;
    el.style.opacity = (.42 + .58 * d).toFixed(3);
    el.style.zIndex = d > .55 ? 4 : 1;
  });
}

/* ── Choreography ── */
const CANCEL = Symbol('cancel');
let RUN = 0;
const wait = (ms, run) => new Promise((res, rej) => setTimeout(() => (run != null && run !== RUN) ? rej(CANCEL) : res(), ms));
const waitFor = (fn, run) => new Promise((res, rej) => { const iv = setInterval(() => { if (run != null && run !== RUN) { clearInterval(iv); rej(CANCEL); } else if (fn()) { clearInterval(iv); res(); } }, 120); });
const later = (fn, ms, run) => setTimeout(() => { if (run == null || run === RUN) fn(); }, ms);
const guard = p => p.catch(e => { if (e !== CANCEL) console.error(e); });

async function assignTask(id, title) {
  const num = S.taskN++;
  const t = {key:`task:${num}`, id:`TASK-${num}`, title, short: title.length > 26 ? title.slice(0, 24) + '…' : title, owner:id, status:'queued', steps:3, updated:'now'};
  TASKS.unshift(t);
  const tv = TaskView(t, {agents:'1 agent', branch: A[id].tool === 'code' ? `fix/${num}` : 'main'});
  openTab(t.key);
  tv.row({agent:'you', kind:'note', icon:'plus', verb:'Created task', arg:t.id, done:true});
  if (ACTIVE.has(S.agents[id].presence) || ATTN.has(S.agents[id].presence)) {
    const w = tv.row({agent:id, icon:'clock', verb:`Queued behind ${A[id].name}’s current work`, ind:'wait'});
    await waitFor(() => !ACTIVE.has(S.agents[id].presence) && !ATTN.has(S.agents[id].presence));
    w.done({verb:'Started', meta:'queue cleared'});
  }
  await taskStart(id, {presence:'thinking', label:'Planning…'});
  tv.status('running');
  const steps = MINI[A[id].tool];
  tv.plan(steps.map(s => ({title:s[0], agent:id})));
  for (let k = 0; k < steps.length; k++) {
    const [title2, icn, doneVerb, meta] = steps[k];
    tv.step(k, 'running');
    setAgent(id, {presence:'tool', tool:A[id].tool, label:title2});
    const row = tv.row({agent:id, icon:icn, verb:title2, ind:TOOLS[A[id].tool].ind});
    await sleep(rand(1800, 2600));
    row.done({verb:doneVerb, meta});
    tv.step(k, 'done');
  }
  tv.status('done');
  tv.row({agent:id, kind:'note', icon:'check', verb:'Completed task', meta:'3 of 3 steps', done:true});
  React.celebrate(id); setAgent(id, {presence:'complete', label:'Task completed'});
  toast({agent:id, kind:'done', html:`<b>${A[id].name}</b> completed ${esc(t.id)}`, sub:title});
  await sleep(2200);
  if (S.agents[id].presence === 'complete') setAgent(id, {presence:'available'});
}
